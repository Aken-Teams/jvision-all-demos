-- 重建客戶終身價值
with orders as (
  select
    customer_id,
    sum(gross_amount - refund_amount)
      as net_revenue
  from warehouse.fact_orders
  where order_state = 'completed'
  group by customer_id
)
select
  c.customer_id,
  coalesce(o.net_revenue, 0)
    as customer_lifetime_value,
  current_timestamp() as lineage_loaded_at
from core.dim_customer c
left join orders o using (customer_id)
