import { Injectable, Logger } from '@nestjs/common';
import { InternalApiClient } from '@jvision/api-client';

@Injectable()
export class OrderHistoryService {
  private readonly logger = new Logger(OrderHistoryService.name);

  constructor(private readonly api: InternalApiClient) {}

  async findRecentOrders(customerId: string) {
    return this.api.get(
      `/commerce-order/v3/customers/${customerId}/orders`,
      {
        params: { days: 90, include: 'payment,shipment' },
        timeout: 2500,
        retry: { attempts: 2, retryOn: [429, 503] },
        headers: { 'x-trace-id': RequestContext.traceId },
        // 不記錄 customerId，避免個資進入應用日誌
        redact: ['request.path.customerId']
      }
    );
  }
}
